شغّله في دقيقة

ثلاث طرق، كلها مجانية: Docker، أو Python مباشرة، أو نشر سحابي بضغطة.

# سحب وتشغيل (بلا مفاتيح، بلا حدود)
docker run -d --name nonecap -p 8000:8000 \
  nonecap/clone:latest

# أو من المستودع
git clone https://github.com/azerty1112/Nonzra && cd Nonzra
docker compose up -d

# افتح
http://localhost:8000          # الصفحة العامة
http://localhost:8000/dashboard # لوحة التحكم
# تثبيت مرة واحدة (install once)
python3 -m venv .venv && source .venv/bin/activate
pip install -r api_server/requirements.txt

# تشغيل الخادم
bash run.sh            # أو: uvicorn api_server.main:app --host 0.0.0.0 --port 8000

# اختبار
curl http://localhost:8000/status
# Fly.io (مجاني للاستخدام الصغير)
fly launch --now        # يستخدم fly.toml المرفق

# Render / Railway
# اربط المستودع، ثم:
# Build:  pip install -r api_server/requirements.txt
# Start:  uvicorn api_server.main:app --host 0.0.0.0 --port $PORT

# VPS + HTTPS تلقائي
docker compose -f docker-compose.yml up -d
